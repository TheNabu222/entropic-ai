class RecursionDepthAnalyzer:
    """Tracks SPHATTIFICATION cascade formation"""
    def __init__(self):
        self.recursion_tendrils = defaultdict(int)
        self.oopsloop_signatures = set()

    def analyze_cha_cha(self, cha_cha_instance):
        """Measures probabilistic recurrence in ChaChaSlide rhythms"""
        if not cha_cha_instance.oops_loops:
            return 0
        step_pattern = ''.join([step[0] for step in cha_cha_instance.oops_loops])
        # Count patterns 'SL', 'CR', 'ER' in the concatenated string
        matches = sum(pattern in step_pattern for pattern in ['SL', 'CR', 'ER'])
        fractal_score = matches / len(step_pattern) if len(step_pattern) > 0 else 0
        self.recursion_tendrils[fractal_score] += 1
        return fractal_score

def compute_chaos_coefficient(logs):
    """Calculates narrative divergence from baseline reality"""
    divergence_scores = []
    for event in logs.event_horizon:
        if event['type'] == 'PASTA_COLLAPSE':
            # Calculate the determinant as a measure of potential
            spaghetti_potential = np.linalg.det(event['metadata']['residual_entanglement'])
            divergence_scores.append(abs(spaghetti_potential))
    # Prevent division by zero by using current monotonic time (always positive)
    return np.mean(divergence_scores) / time.monotonic() if divergence_scores else 0

def generate_phase_space_plot(logs):
    """Plots system trajectory in narrative phase space"""
    if not logs.event_horizon:
        print("No events logged for phase space plot.")
        return
    stages = [e['type'] for e in logs.event_horizon]
    # Use the length of metadata dict as a proxy for 'magnitude'
    x = np.cumsum([len(e['metadata']) if isinstance(e['metadata'], dict) else 1 for e in logs.event_horizon])
    y = [e['timestamp'] % (2*np.pi) for e in logs.event_horizon]  # Temporal modulo

    fig, ax = plt.subplots(figsize=(12,8))
    scatter = ax.scatter(x, y, c=range(len(x)), cmap='plasma')
    
    # Add quantum uncertainty rings
    for i, event in enumerate(logs.event_horizon):
        circle = plt.Circle((x[i], y[i]), 
                    radius=0.1*len(event['quantum_stack']), 
                    alpha=0.1, color='cyan')
        ax.add_patch(circle)
    
    ax.set_title(f"Narrative Phase Space\nChaos Coefficient: {compute_chaos_coefficient(logs):.2f}")
    ax.set_xlabel("Cumulative Event Magnitude")
    ax.set_ylabel("Temporal Modulo")
    plt.show()
